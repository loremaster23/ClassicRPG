5. SERVIDOR ACTUAL
========================================

Archivo:
scripts/servidor/Servidor.gd

Código:

extends Node

const PUERTO := 9000
const MAX_JUGADORES := 2

var servidor := WebSocketMultiplayerPeer.new()
var jugadores := 0

func _ready() -> void:
    servidor.peer_connected.connect(_al_conectar)
    servidor.peer_disconnected.connect(_al_desconectar)

    var error := servidor.create_server(PUERTO, "*")

    if error != OK:
        push_error("No se pudo iniciar el servidor: %s" % error)
        return

    print("Servidor iniciado. Máximo de jugadores: ", MAX_JUGADORES)

func _process(_delta: float) -> void:
    servidor.poll()

    while servidor.get_available_packet_count() > 0:
        var id := servidor.get_packet_peer()
        var mensaje := servidor.get_packet()
        print("Jugador ", id, ": ", mensaje.get_string_from_utf8())

func _al_conectar(id: int) -> void:
    jugadores += 1
    print("Jugador conectado: ", id, " | ", jugadores, "/", MAX_JUGADORES)

    if jugadores >= MAX_JUGADORES:
        servidor.refuse_new_connections = true
        print("Servidor lleno. No acepta más conexiones.")

func _al_desconectar(id: int) -> void:
    jugadores -= 1
    print("Jugador desconectado: ", id, " | ", jugadores, "/", MAX_JUGADORES)
    servidor.refuse_new_connections = false

func _exit_tree() -> void:
    servidor.close()

6. CLIENTE DE PRUEBA ACTUAL
========================================

Archivo:
scripts/cliente/ClientePrueba.gd

Código:

extends Node

const DIRECCION := "ws://127.0.0.1:9000"

var cliente := WebSocketMultiplayerPeer.new()
var mensaje_enviado := false
var tiempo := 0.0

func _ready() -> void:
    var error := cliente.create_client(DIRECCION)

    if error != OK:
        push_error("Error al conectar: %s" % error)
        get_tree().quit()
        return

    print("Conectando con el servidor...")

func _process(delta: float) -> void:
    cliente.poll()
    tiempo += delta

    var estado := cliente.get_connection_status()

    if estado == MultiplayerPeer.CONNECTION_CONNECTED and not mensaje_enviado:
        var error := cliente.put_packet("Hola desde el cliente".to_utf8_buffer())

        if error == OK:
            print("Conectado. Mensaje enviado correctamente.")
            mensaje_enviado = true

    if estado == MultiplayerPeer.CONNECTION_DISCONNECTED and tiempo > 10.0:
        push_error("No se pudo mantener la conexión.")
        get_tree().quit()

    if mensaje_enviado and tiempo > 60.0:
        print("Finalizando cliente de prueba.")
        cliente.close()
        get_tree().quit()

func _exit_tree() -> void:
    cliente.close()

Escena:
escenas/cliente/ClientePrueba.tscn

El cliente envía un mensaje al servidor y finaliza después de 60 segundos.
